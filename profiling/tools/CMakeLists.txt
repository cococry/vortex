cmake_minimum_required(VERSION 3.16)
project(vortex_profiling_tools LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_BUILD_TYPE Release)
if(NOT TRACY_SOURCE)
  message(FATAL_ERROR "Pass -DTRACY_SOURCE=/path/to/patched/tracy")
endif()
# Only the server and two command-line tools. No vendor.cmake, GUI, network
# dependencies, disassembler, or parallel-sort library.
add_library(TracyCapstone INTERFACE)
add_library(libzstd INTERFACE)
target_link_libraries(libzstd INTERFACE zstd)
add_library(ppq INTERFACE)
add_library(PPQSort::PPQSort ALIAS ppq)
include(${TRACY_SOURCE}/cmake/server.cmake)
add_executable(tracy-csvexport ${TRACY_SOURCE}/csvexport/src/csvexport.cpp ${TRACY_SOURCE}/getopt/getopt.c)
add_executable(tracy-capture ${TRACY_SOURCE}/capture/src/capture.cpp)
foreach(tool tracy-csvexport tracy-capture)
  target_link_libraries(${tool} PRIVATE TracyServer pthread dl)
endforeach()
